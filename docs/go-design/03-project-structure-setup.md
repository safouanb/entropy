# Go Backend Project Structure & Setup

## Overview

This document defines the complete project structure, Gin routing, ConnectRPC setup, middleware, and build automation for the PyRecycle Heat Go backend.

**Framework:** Gin + ConnectRPC  
**Build Tool:** Make + Go modules  
**Architecture:** Clean Architecture / Hexagonal

## main.go (Application Entry Point)

```go
package main

import (
    "context"
    "database/sql"
    "fmt"
    "log/slog"
    "net/http"
    "os"
    "os/signal"
    "syscall"
    "time"

    "github.com/gin-gonic/gin"
    "github.com/pyrecycleheat/backend/internal/config"
    "github.com/pyrecycleheat/backend/internal/database"
    "github.com/pyrecycleheat/backend/internal/engine"
    "github.com/pyrecycleheat/backend/internal/handler/connectrpc"
    "github.com/pyrecycleheat/backend/internal/handler/rest"
    "github.com/pyrecycleheat/backend/internal/router"
    "github.com/pyrecycleheat/backend/internal/service"
)

func main() {
    // Load configuration
    cfg, err := config.Load()
    if err != nil {
        slog.Error("failed to load config", "error", err)
        os.Exit(1)
    }

    // Setup structured logging
    logger := setupLogger(cfg.LogLevel)
    slog.SetDefault(logger)

    slog.Info("starting pyrecycleheat backend", "version", cfg.Version, "env", cfg.Environment)

    // Connect to database
    db, err := database.Open(cfg.DatabaseURL)
    if err != nil {
        slog.Error("failed to open database", "error", err)
        os.Exit(1)
    }
    defer db.Close()

    // Run migrations
    if err := database.RunMigrations(db, cfg.MigrationsPath); err != nil {
        slog.Error("failed to run migrations", "error", err)
        os.Exit(1)
    }

    slog.Info("database migrations applied successfully")

    // Initialize services
    queries := database.New(db)
    predictionEngine := engine.NewPredictionEngine()

    districtHeatingSvc := service.NewDistrictHeatingService(db, queries, logger)
    predictionSvc := service.NewPredictionService(db, queries, predictionEngine, logger)

    // Setup Gin router (REST API for compatibility)
    ginRouter := setupGinRouter(cfg, districtHeatingSvc, predictionSvc, logger)

    // Setup ConnectRPC mux (RPC API)
    connectMux := setupConnectRPCMux(cfg, districtHeatingSvc, predictionSvc, logger)

    // Create main HTTP server that handles both REST and RPC
    mainHandler := http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        // Route ConnectRPC requests to connect mux
        if r.Header.Get("Content-Type") == "application/proto" ||
           r.Header.Get("Content-Type") == "application/json" && r.URL.Path[:4] == "/rpc" {
            connectMux.ServeHTTP(w, r)
            return
        }
        
        // Route REST requests to Gin
        ginRouter.ServeHTTP(w, r)
    })

    server := &http.Server{
        Addr:         cfg.ServerAddress,
        Handler:      mainHandler,
        ReadTimeout:  15 * time.Second,
        WriteTimeout: 15 * time.Second,
        IdleTimeout:  60 * time.Second,
    }

    // Start server in goroutine
    go func() {
        slog.Info("server starting", "address", cfg.ServerAddress)
        if err := server.ListenAndServe(); err != nil && err != http.ErrServerClosed {
            slog.Error("server error", "error", err)
            os.Exit(1)
        }
    }()

    // Graceful shutdown
    quit := make(chan os.Signal, 1)
    signal.Notify(quit, syscall.SIGINT, syscall.SIGTERM)
    <-quit

    slog.Info("shutting down server...")

    ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
    defer cancel()

    if err := server.Shutdown(ctx); err != nil {
        slog.Error("server forced to shutdown", "error", err)
    }

    slog.Info("server exited")
}

func setupLogger(level string) *slog.Logger {
    var logLevel slog.Level
    switch level {
    case "debug":
        logLevel = slog.LevelDebug
    case "info":
        logLevel = slog.LevelInfo
    case "warn":
        logLevel = slog.LevelWarn
    case "error":
        logLevel = slog.LevelError
    default:
        logLevel = slog.LevelInfo
    }

    handler := slog.NewJSONHandler(os.Stdout, &slog.HandlerOptions{
        Level: logLevel,
    })

    return slog.New(handler)
}

func setupGinRouter(cfg *config.Config, districtSvc *service.DistrictHeatingService, predictionSvc *service.PredictionService, logger *slog.Logger) *gin.Engine {
    if cfg.Environment == "production" {
        gin.SetMode(gin.ReleaseMode)
    }

    r := gin.New()
    
    // Setup middleware
    r.Use(ginLogger(logger))
    r.Use(gin.Recovery())
    r.Use(corsMiddleware(cfg.AllowedOrigins))

    // Initialize REST handlers
    districtHandler := rest.NewDistrictHeatingHandler(districtSvc, logger)
    predictionHandler := rest.NewPredictionHandler(predictionSvc, logger)

    // Register REST routes
    router.RegisterRESTRoutes(r, districtHandler, predictionHandler)

    return r
}

func setupConnectRPCMux(cfg *config.Config, districtSvc *service.DistrictHeatingService, predictionSvc *service.PredictionService, logger *slog.Logger) *http.ServeMux {
    mux := http.NewServeMux()

    // Initialize ConnectRPC handlers
    districtHandler := connectrpc.NewDistrictHeatingHandler(districtSvc, logger)
    predictionHandler := connectrpc.NewPredictionHandler(predictionSvc, logger)

    // Register ConnectRPC services
    router.RegisterConnectRPCServices(mux, districtHandler, predictionHandler)

    return mux
}

func ginLogger(logger *slog.Logger) gin.HandlerFunc {
    return func(c *gin.Context) {
        start := time.Now()
        path := c.Request.URL.Path
        raw := c.Request.URL.RawQuery

        c.Next()

        latency := time.Since(start)
        statusCode := c.Writer.Status()
        method := c.Request.Method
        clientIP := c.ClientIP()

        if raw != "" {
            path = path + "?" + raw
        }

        logger.Info("request",
            slog.Int("status", statusCode),
            slog.String("method", method),
            slog.String("path", path),
            slog.String("ip", clientIP),
            slog.Duration("latency", latency),
        )
    }
}

func corsMiddleware(allowedOrigins []string) gin.HandlerFunc {
    return func(c *gin.Context) {
        origin := c.Request.Header.Get("Origin")
        
        // Check if origin is allowed
        allowed := false
        for _, allowedOrigin := range allowedOrigins {
            if origin == allowedOrigin {
                allowed = true
                break
            }
        }

        if allowed {
            c.Writer.Header().Set("Access-Control-Allow-Origin", origin)
            c.Writer.Header().Set("Access-Control-Allow-Credentials", "true")
            c.Writer.Header().Set("Access-Control-Allow-Methods", "GET, POST, PUT, DELETE, OPTIONS")
            c.Writer.Header().Set("Access-Control-Allow-Headers", "Content-Type, Authorization")
        }

        if c.Request.Method == "OPTIONS" {
            c.AbortWithStatus(204)
            return
        }

        c.Next()
    }
}
```

```go
// RegisterRESTRoutes registers all REST API routes
func RegisterRESTRoutes(r *gin.Engine, districtHandler *rest.DistrictHeatingHandler, predictionHandler *rest.PredictionHandler) {
    // Health check
    r.GET("/health", func(c *gin.Context) {
        c.JSON(200, gin.H{"status": "healthy"})

    // API v1 group
    v1 := r.Group("/api/v1")
    {

        registerDistrictHeatingRoutes(v1, districtHandler)

        // Prediction routes
        registerPredictionRoutes(v1, predictionHandler)
```

## ConnectRPC Router Setup (router/connectrpc.go)

```go
package router

import (
    "net/http"

    "connectrpc.com/connect"
    "github.com/pyrecycleheat/backend/internal/handler/connectrpc"
    v1 "github.com/pyrecycleheat/api/gen/go/pyrecycleheat/v1"
    "github.com/pyrecycleheat/api/gen/go/pyrecycleheat/v1/pyrecycleheatv1connect"
)

// RegisterConnectRPCServices registers all ConnectRPC services
func RegisterConnectRPCServices(mux *http.ServeMux, districtHandler *connectrpc.DistrictHeatingHandler, predictionHandler *connectrpc.PredictionHandler) {
    // Setup interceptors
    interceptors := connect.WithInterceptors(
        loggingInterceptor(),
        errorInterceptor(),
    )

    // Register District Heating Service
    districtPath, districtHandlerFunc := pyrecycleheatv1connect.NewDistrictHeatingServiceHandler(
        districtHandler,
        interceptors,
    )
    mux.Handle(districtPath, districtHandlerFunc)

    // Register Prediction Service
    predictionPath, predictionHandlerFunc := pyrecycleheatv1connect.NewPredictionServiceHandler(
        predictionHandler,
        interceptors,
    )
    mux.Handle(predictionPath, predictionHandlerFunc)
}

// loggingInterceptor logs all RPC calls
func loggingInterceptor() connect.UnaryInterceptorFunc {
    return func(next connect.UnaryFunc) connect.UnaryFunc {
        return func(ctx context.Context, req connect.AnyRequest) (connect.AnyResponse, error) {
            start := time.Now()
            
            slog.InfoContext(ctx, "rpc call",
                slog.String("procedure", req.Spec().Procedure),
            )

            res, err := next(ctx, req)

            latency := time.Since(start)
            
            if err != nil {
                slog.ErrorContext(ctx, "rpc error",
                    slog.String("procedure", req.Spec().Procedure),
                    slog.Duration("latency", latency),
                    slog.String("error", err.Error()),
                )
            } else {
                slog.InfoContext(ctx, "rpc success",
                    slog.String("procedure", req.Spec().Procedure),
                    slog.Duration("latency", latency),
                )
            }

            return res, err
        }
    }
}

// errorInterceptor converts service errors to Connect errors
func errorInterceptor() connect.UnaryInterceptorFunc {
    return func(next connect.UnaryFunc) connect.UnaryFunc {
        return func(ctx context.Context, req connect.AnyRequest) (connect.AnyResponse, error) {
            res, err := next(ctx, req)
            if err != nil {
                // Convert custom errors to Connect codes
                return res, convertToConnectError(err)
            }
            return res, nil
        }
    }
}

func convertToConnectError(err error) error {
    switch e := err.(type) {
    case *service.NotFoundError:
        return connect.NewError(connect.CodeNotFound, err)
    case *service.ValidationError:
        return connect.NewError(connect.CodeInvalidArgument, err)
    case *service.ConflictError:
        return connect.NewError(connect.CodeAlreadyExists, err)
    default:
        return connect.NewError(connect.CodeInternal, err)
    }
}
```

## Configuration Management (config/config.go)

```go
package config

import (
    "fmt"
    "os"

    "github.com/spf13/viper"
)

type Config struct {
    Environment    string   `mapstructure:"environment"`
    Version        string   `mapstructure:"version"`
    ServerAddress  string   `mapstructure:"server_address"`
    DatabaseURL    string   `mapstructure:"database_url"`
    MigrationsPath string   `mapstructure:"migrations_path"`
    LogLevel       string   `mapstructure:"log_level"`
    AllowedOrigins []string `mapstructure:"allowed_origins"`
}

// Load loads configuration from file and environment variables
func Load() (*Config, error) {
    viper.SetConfigName("config")
    viper.SetConfigType("yaml")
    viper.AddConfigPath("./internal/config")
    viper.AddConfigPath(".")

    // Environment variable overrides
    viper.SetEnvPrefix("PYRECYCLEHEAT")
    viper.AutomaticEnv()

    // Set defaults
    viper.SetDefault("environment", "development")
    viper.SetDefault("version", "1.0.0")
    viper.SetDefault("server_address", ":8080")
    viper.SetDefault("database_url", "sqlite://./district_heating.db")
    viper.SetDefault("migrations_path", "./internal/database/migrations")
    viper.SetDefault("log_level", "info")
    viper.SetDefault("allowed_origins", []string{
        "http://localhost:5173",
        "http://localhost:3000",
        "https://pyrecycleheat.vercel.app",
    })

    if err := viper.ReadInConfig(); err != nil {
        if _, ok := err.(viper.ConfigFileNotFoundError); !ok {
            return nil, fmt.Errorf("read config: %w", err)
        }
        // Config file not found, use defaults
    }

    var cfg Config
    if err := viper.Unmarshal(&cfg); err != nil {
        return nil, fmt.Errorf("unmarshal config: %w", err)
    }

    // Override with environment variables
    if dbURL := os.Getenv("DATABASE_URL"); dbURL != "" {
        cfg.DatabaseURL = dbURL
    }

    return &cfg, nil
}
```

## Dockerfile (Multi-Stage Build)

```dockerfile
# Build stage
FROM golang:1.23-alpine AS builder

# Install build dependencies
RUN apk add --no-cache git gcc musl-dev sqlite-dev

WORKDIR /build

# Copy go mod files
COPY go.mod go.sum ./
RUN go mod download

# Copy source code
COPY . .

# Build binary
RUN CGO_ENABLED=1 GOOS=linux go build -ldflags="-s -w" -o /dist/server ./cmd/server

# Runtime stage
FROM alpine:latest

# Install runtime dependencies
RUN apk --no-cache add ca-certificates sqlite-libs

WORKDIR /app

# Copy binary from builder
COPY --from=builder /dist/server ./server

# Copy configuration
COPY internal/config/config.yaml ./config.yaml

# Copy migrations
COPY internal/database/migrations ./migrations

# Expose port
EXPOSE 8080

# Health check
HEALTHCHECK --interval=30s --timeout=3s --start-period=5s --retries=3 \
    CMD wget --no-verbose --tries=1 --spider http://localhost:8080/health || exit 1

# Run server
ENTRYPOINT ["./server"]
```
